# 1. Dependências de build
sudo apt update
sudo apt install -y cmake git build-essential libgl1-mesa-dev libx11-dev

# 2. Baixa e compila raylib do fonte (versão compatível com 4 args)
git clone --depth 1 --branch 4.5.0 https://github.com/raysan5/raylib.git
cd raylib/src
make PLATFORM=PLATFORM_DESKTOP
sudo make install
cd ../..

# 3. Agora compila o projeto
sh build.sh
